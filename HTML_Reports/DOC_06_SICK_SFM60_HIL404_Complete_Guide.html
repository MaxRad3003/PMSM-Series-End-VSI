הקוד מפצל את חבילת 10 הבייטים, מחלץ את ביטי ה-Singleturn וה-Multiturn, ומחשב מהירות מסוננת ב-RPM עם מנגנון Deadband:

// 1. חיבור 32 ביט המיקום (MSB b5 -> LSB b8) uint32_t pos_raw = ((uint32_t)b5 << 24) | ((uint32_t)b6 << 16) | ((uint32_t)b7 << 8) | (uint32_t)b8; // 2. חילוץ Singleturn (15 ביטים תחתונים: 0 עד 32,767) uint32_t st_steps = pos_raw & 0x7FFF; double raw_deg = ((double)st_steps / 32768.0) * 360.0; // 3. חילוץ Multiturn (ספירת 4,096 סיבובים שלמים) uint32_t multiturn_count = (pos_raw >> 15) & 0xFFF; // 4. חישוב זווית חשמלית Theta_e עבור 4 זוגות קטבים (POLE_PAIRS = 4) double theta_e = fmod(4.0 * (filtered_deg * (PI / 180.0)), 2.0 * PI); // 5. חישוב מהירות מדויקת ב-RPM עם Deadband (0.00 RPM בעמידה במקום) double delta_deg = filtered_deg - prev_pos_deg; if (fabs(delta_deg) < 0.005) { filtered_rpm = 0.0; // עצירה מלאה } else { double raw_rpm = (delta_deg / 360.0) / 0.0001 * 60.0; filtered_rpm += 0.02 * (raw_rpm - filtered_rpm); }
